df = pd.read_csv('sales_data.csv', parse_dates=['date'])

# Feature engineering
df['revenue']  = df['units'] * df['unit_price'] * (1 - df['discount_pct'] / 100)
df['cost']     = df['units'] * df['cost_per_unit']
df['profit']   = df['revenue'] - df['cost']
df['margin']   = df['profit'] / df['revenue'] * 100
df['month']    = df['date'].dt.to_period('M')
df['quarter']  = df['date'].dt.to_period('Q')
df['month_no'] = df['date'].dt.month

# Check for nulls
print("Null values:\n", df.isnull().sum())
print(f"\nDate range: {df['date'].min().date()} → {df['date'].max().date()}")
print(f"Total rows: {len(df)}")

monthly = df.groupby('month_no')['revenue'].sum().reset_index()
month_labels = ['Jan','Feb','Mar','Apr','May','Jun',
                'Jul','Aug','Sep','Oct','Nov','Dec']

fig, ax = plt.subplots(figsize=(12, 4))
ax.fill_between(monthly['month_no'], monthly['revenue'],
                alpha=0.15, color='#2563EB')
ax.plot(monthly['month_no'], monthly['revenue'],
        color='#2563EB', linewidth=2.5, marker='o', markersize=6)

for x, y in zip(monthly['month_no'], monthly['revenue']):
    ax.annotate(f"${y/1000:.1f}K", (x, y),
                textcoords="offset points", xytext=(0, 10),
                ha='center', fontsize=8.5, color='#374151')

ax.set_xticks(range(1, 13))
ax.set_xticklabels(month_labels)
ax.yaxis.set_major_formatter(mticker.FuncFormatter(lambda x, _: f"${x/1000:.0f}K"))
ax.set_title("Monthly Revenue Trend — FY 2024", fontsize=14, fontweight='bold', pad=15)
ax.set_xlabel("Month"); ax.set_ylabel("Revenue")
plt.tight_layout()
plt.savefig('outputs/01_monthly_trend.png', bbox_inches='tight')
plt.show()
print("✅ Saved: 01_monthly_trend.png")

cat_rev = df.groupby('category')['revenue'].sum().sort_values(ascending=False)

fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(12, 5))

# Bar chart
bars = ax1.barh(cat_rev.index, cat_rev.values,
                color=[COLORS[c] for c in cat_rev.index], edgecolor='none', height=0.55)
for bar, val in zip(bars, cat_rev.values):
    ax1.text(val + 200, bar.get_y() + bar.get_height()/2,
             f"${val:,.0f}", va='center', fontsize=9.5)
ax1.set_title("Revenue by Category", fontsize=13, fontweight='bold')
ax1.xaxis.set_major_formatter(mticker.FuncFormatter(lambda x, _: f"${x/1000:.0f}K"))
ax1.invert_yaxis()

# Pie chart
wedges, texts, autotexts = ax2.pie(
    cat_rev.values,
    labels=cat_rev.index,
    colors=[COLORS[c] for c in cat_rev.index],
    autopct='%1.1f%%',
    startangle=140,
    wedgeprops=dict(edgecolor='white', linewidth=2),
    pctdistance=0.75
)
for t in autotexts:
    t.set_fontsize(9)
ax2.set_title("Category Revenue Share", fontsize=13, fontweight='bold')

plt.tight_layout()

reg = df.groupby('region').agg(
    Revenue=('revenue', 'sum'),
    Profit=('profit', 'sum'),
    Units=('units', 'sum'),
    Margin=('margin', 'mean')
).sort_values('Revenue', ascending=False)

fig, axes = plt.subplots(1, 3, figsize=(14, 5))

# Revenue
axes[0].bar(reg.index, reg['Revenue'],
            color=REGION_COLORS, edgecolor='none', width=0.55)
axes[0].set_title("Revenue by Region", fontweight='bold')
axes[0].yaxis.set_major_formatter(mticker.FuncFormatter(lambda x,_: f"${x/1000:.0f}K"))
for i, (r, v) in enumerate(zip(reg.index, reg['Revenue'])):
    axes[0].text(i, v + 200, f"${v/1000:.1f}K", ha='center', fontsize=8.5)

# Units
axes[1].bar(reg.index, reg['Units'],
            color=REGION_COLORS, edgecolor='none', width=0.55)
axes[1].set_title("Units Sold by Region", fontweight='bold')
for i, (r, v) in enumerate(zip(reg.index, reg['Units'])):
    axes[1].text(i, v + 5, str(int(v)), ha='center', fontsize=8.5)

# Margin
bars = axes[2].bar(reg.index, reg['Margin'],
                   color=REGION_COLORS, edgecolor='none', width=0.55)
axes[2].set_title("Avg Gross Margin % by Region", fontweight='bold')

for i, (r, v) in enumerate(zip(reg.index, reg['Margin'])):
    axes[2].text(i, v + 0.3, f"{v:.1f}%", ha='center', fontsize=8.5)
